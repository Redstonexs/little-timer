cmake_minimum_required(VERSION 3.16)
project(LittleTimer VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(timer_core src/timer.cpp src/sound.cpp)
target_include_directories(timer_core PUBLIC src)
if(MSVC)
    set_property(TARGET timer_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(timer_core PRIVATE /utf-8 /W4)
else()
    target_compile_options(timer_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(timer_tests tests/timer_tests.cpp)
    target_link_libraries(timer_tests PRIVATE timer_core)
    if(MSVC)
        set_property(TARGET timer_tests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        target_compile_options(timer_tests PRIVATE /utf-8)
    elseif(MINGW)
        target_link_options(timer_tests PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME timer_core COMMAND timer_tests)
endif()

if(WIN32)
    enable_language(RC)
    add_executable(LittleTimer WIN32 src/main.cpp src/app.cpp src/ui.cpp src/settings_dialog.cpp src/win_audio.cpp resources/app.rc)
    target_link_libraries(LittleTimer PRIVATE timer_core gdiplus winmm comctl32 gdi32 user32 shell32)
    target_include_directories(LittleTimer PRIVATE resources)
    target_compile_definitions(LittleTimer PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0601 WINVER=0x0601)
    if(MSVC)
        set_property(TARGET LittleTimer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        target_compile_options(LittleTimer PRIVATE /utf-8 /W4)
        target_link_options(LittleTimer PRIVATE /SUBSYSTEM:WINDOWS,6.01)
    else()
        target_compile_options(LittleTimer PRIVATE -Wall -Wextra -Wpedantic)
        target_link_options(LittleTimer PRIVATE -municode -static -static-libgcc -static-libstdc++
            -Wl,--major-os-version,6,--minor-os-version,1,--major-subsystem-version,6,--minor-subsystem-version,1
            -Wl,--dynamicbase,--nxcompat,--no-insert-timestamp)
    endif()
endif()
